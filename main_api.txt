# api/main_api.py - Backend API NOKIROVA 🌐⚡ (Production Render)

import sys
import os

# Ajouter le dossier parent au path
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from fastapi import FastAPI, UploadFile, File, Form, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import FileResponse
from pydantic import BaseModel
import shutil
import uuid

# Imports NOKIROVA (sans interface graphique)
from ia_handler import demander_ia, expliquer_simplement, creer_resume
from document_parser import lire_document
from audio_generator import generer_audio
from exercice_generator import generer_qcm, generer_questions_cours, generer_exercices_examen
from ocr_handler import lire_image_et_expliquer
from export_pdf import exporter_en_pdf
import database as db

# Initialiser
db.init_db()
db.maj_streak()

# ═══════════════════════════════════════════
# 🚀 CRÉATION DE L'API
# ═══════════════════════════════════════════
app = FastAPI(
    title="NOKIROVA API",
    description="🌸 Ton Prof Intelligent Personnel",
    version="1.0.0"
)

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

UPLOAD_FOLDER = "uploads"
OUTPUT_FOLDER = "outputs"
os.makedirs(UPLOAD_FOLDER, exist_ok=True)
os.makedirs(OUTPUT_FOLDER, exist_ok=True)

sessions = {}


# ═══════════════════════════════════════════
# 📋 MODÈLES
# ═══════════════════════════════════════════
class QuestionLibre(BaseModel):
    question: str


class GenerationRequest(BaseModel):
    session_id: str
    nombre: int = 5


class ExamenRequest(BaseModel):
    session_id: str
    nombre: int = 3
    niveau: str = "intermediaire"


class AudioRequest(BaseModel):
    texte: str
    voix: str = "jeune_femme"


# ═══════════════════════════════════════════
# 🏠 ACCUEIL
# ═══════════════════════════════════════════
@app.get("/")
async def accueil():
    return {
        "message": "🌸 Bienvenue dans l'API NOKIROVA !",
        "version": "1.0.0",
        "status": "✅ Opérationnel",
        "docs": "/docs"
    }


@app.get("/health")
async def health():
    """Endpoint de santé pour Render"""
    return {"status": "healthy"}


# ═══════════════════════════════════════════
# 📥 IMPORTER UN COURS
# ═══════════════════════════════════════════
@app.post("/api/import")
async def importer_cours(fichier: UploadFile = File(...)):
    try:
        session_id = str(uuid.uuid4())
        chemin = os.path.join(UPLOAD_FOLDER, f"{session_id}_{fichier.filename}")

        with open(chemin, "wb") as f:
            shutil.copyfileobj(fichier.file, f)

        contenu = lire_document(chemin)

        if not contenu or contenu.startswith("❌"):
            raise HTTPException(status_code=400, detail="Impossible de lire le fichier")

        matiere = db.sauvegarder_cours(fichier.filename, contenu)

        sessions[session_id] = {
            "contenu": contenu,
            "nom": fichier.filename,
            "matiere": matiere
        }

        return {
            "success": True,
            "session_id": session_id,
            "nom": fichier.filename,
            "matiere": matiere,
            "taille": len(contenu),
            "apercu": contenu[:500] + "..." if len(contenu) > 500 else contenu,
            "message": f"✅ Cours chargé ! Matière : {matiere}"
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Erreur : {str(e)}")


# ═══════════════════════════════════════════
# 📝 RÉSUMÉ
# ═══════════════════════════════════════════
@app.post("/api/resume/{session_id}")
async def creer_resume_endpoint(session_id: str):
    if session_id not in sessions:
        raise HTTPException(status_code=404, detail="Session introuvable")

    contenu = sessions[session_id]["contenu"]
    resume = creer_resume(contenu[:5000])
    db.ajouter_xp(5)

    return {
        "success": True,
        "resume": resume,
        "matiere": sessions[session_id]["matiere"]
    }

# ═══════════════════════════════════════════
# 🎯 QCM
# ═══════════════════════════════════════════

async def generer_qcm_endpoint(req: GenerationRequest):

# ═══════════════════════════════════════════
# 💬 CHAT
# ═══════════════════════════════════════════
@app.post("/api/chat")
async def chat_endpoint(req: QuestionLibre):
    reponse = demander_ia(req.question)
    db.incrementer_stat("questions_posees")
    db.ajouter_xp(5)
    db.sauvegarder_historique("chat", req.question, reponse)

    return {"success": True, "reponse": reponse}


# ═══════════════════════════════════════════
# 🎧 AUDIO
# ═══════════════════════════════════════════
@app.post("/api/audio")
async def creer_audio_endpoint(req: AudioRequest):
    nom_fichier = f"audio_{uuid.uuid4().hex[:8]}.mp3"
    chemin = generer_audio(req.texte, nom_fichier, req.voix)

    if not chemin:
        raise HTTPException(status_code=500, detail="Erreur création audio")

    db.incrementer_stat("audios_crees")
    db.ajouter_xp(8)

    return {
        "success": True,
        "fichier": nom_fichier,
        "url": f"/api/download/{nom_fichier}"
    }


# ═══════════════════════════════════════════
# 📸 OCR (désactivé en ligne)
# ═══════════════════════════════════════════
@app.post("/api/ocr")
async def ocr_endpoint(image: UploadFile = File(...)):
    try:
        chemin = os.path.join(UPLOAD_FOLDER, f"img_{uuid.uuid4().hex[:8]}_{image.filename}")
        with open(chemin, "wb") as f:
            shutil.copyfileobj(image.file, f)

        resultat = lire_image_et_expliquer(chemin)
        db.ajouter_xp(15)

        return {"success": True, "resultat": resultat}
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Erreur OCR : {str(e)}")


# ═══════════════════════════════════════════
# 🏆 PROFIL

@app.get("/api/profil")
async def profil_endpoint():
    stats = db.get_stats()
    badges = db.lister_badges()
    titre = db.get_niveau_titre(stats["niveau"])

    return {
        "success": True,
        "stats": stats,
        "titre": titre,
        "badges": [{"nom": b[0], "description": b[1], "emoji": b[2]} for b in badges]
    }


# ═══════════════════════════════════════════
# 📄 PDF
# ═══════════════════════════════════════════
@app.post("/api/pdf")
async def exporter_pdf_endpoint(
    contenu: str = Form(...),
    titre: str = Form("Document"),
    nom: str = Form("document")
):
    from datetime import datetime
    date_simple = datetime.now().strftime('%d-%m-%Y')
    nom_fichier = f"NOKIROVA_{nom}_{date_simple}.pdf"

    resultat = exporter_en_pdf(contenu, titre, nom_fichier)

    return {"success": True, "resultat": resultat}


# ═══════════════════════════════════════════
# 📥 TÉLÉCHARGEMENT
# ═══════════════════════════════════════════
@app.get("/api/download/{nom_fichier}")
async def telecharger(nom_fichier: str):
    chemin = os.path.join(OUTPUT_FOLDER, nom_fichier)
    if not os.path.exists(chemin):
        raise HTTPException(status_code=404, detail="Fichier introuvable")
    return FileResponse(chemin)
